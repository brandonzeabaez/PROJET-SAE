Description détaillée

Contrôleur principal de l'authentification et des mots de passe.

Il gère les actions de connexion, déconnexion, mot de passe oublié et réinitialisation du mot de passe.

App\Controllers\Login::resetPassword ( )
private

Traite la saisie du nouveau mot de passe via le token reçu par e-mail.

Définition à la ligne 213 du fichier Login.php.

213 : void
214 {
215 // code secret dans l'adresse
216 $token = (string) filter_input(INPUT_GET, 'token');
217 $userModel = new User();
218 $user = $userModel->findByResetToken($token);
219
220 // lien valide et formulaire envoyé : on vérifie le nouveau mdp
221 if ($user !== null && $_SERVER['REQUEST_METHOD'] === 'POST') {
222 $password = (string) filter_input(INPUT_POST, 'password');
223 $confirm = (string) filter_input(INPUT_POST, 'password_confirm');
224
225 $error = '';
226 if (strlen($password) < 8) {
227 $error = 'Le mot de passe doit faire au moins 8 caractères.';
228 } elseif (!preg_match('/[A-Z]/', $password) || !preg_match('/[0-9]/', $password) || !preg_match('/[@$!%*?&]/', $password)) {
229 // une majuscule, un chiffre et un caractère spécial, comme à l'inscription
230 $error = 'Le mot de passe doit contenir une majuscule, un chiffre et un caractère spécial (@ $ ! % * ? &).';
231 } elseif ($password !== $confirm) {
232 $error = 'Les mots de passe ne correspondent pas.';
233 }
234
235 if ($error !== '') {
236 $_SESSION['error'] = $error;
237 $this->redirect('/reinitialiser?token=' . urlencode($token));
238 }
239
240 // on enregistre le mdp haché, jamais en clair
241 $userModel->updatePassword((int) $user['id_user'], password_hash($password, PASSWORD_DEFAULT));
242
243 $_SESSION['message'] = 'Mot de passe modifié, vous pouvez vous connecter.';
244 $this->redirect('/connexion');
245 }
246
247 $error = $_SESSION['error'] ?? '';
248 unset($_SESSION['error']);
249
250 // code inconnu ou expiré : pas de formulaire
251 if ($user === null) {
252 $error = 'Lien invalide ou expiré.';
253 $token = '';
254 }
255
256 $this->render('Auth/resetPassword', [
257 'title' => 'Nouveau mot de passe',
258 'description' => 'Choisissez un nouveau mot de passe.',
259 'error' => $error,
260 'token' => $token,
261 ]);
262 }
