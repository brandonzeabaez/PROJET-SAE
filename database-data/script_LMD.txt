USE sae30392_mysql;

DROP TABLE IF EXISTS Reserve, Token, Utilisateur, Administrateur, Annonce, Profil, Departement, Site, Categorie, Hierarchie, Hierachie;

CREATE TABLE Hierarchie(
   id_hierarchie INT AUTO_INCREMENT,
   role VARCHAR(50) NOT NULL,
   PRIMARY KEY(id_hierarchie)
);

CREATE TABLE Categorie(
   id_categorie INT AUTO_INCREMENT,
   categorie VARCHAR(50) NOT NULL,
   PRIMARY KEY(id_categorie)
);

CREATE TABLE Site(
   id_site INT AUTO_INCREMENT,
   nom_site VARCHAR(50) NOT NULL,
   PRIMARY KEY(id_site)
);

CREATE TABLE Departement(
   id_departement INT AUTO_INCREMENT,
   nom_departement VARCHAR(50) NOT NULL,
   PRIMARY KEY(id_departement)
);

CREATE TABLE Profil(
   id_Utilisateur INT AUTO_INCREMENT,
   email VARCHAR(255) NOT NULL UNIQUE,
   mot_de_passe VARCHAR(255) NOT NULL,
   email_verifiee BOOLEAN NOT NULL DEFAULT FALSE,
   id_hierarchie INT NOT NULL,
   PRIMARY KEY(id_Utilisateur),
   FOREIGN KEY(id_hierarchie) REFERENCES Hierarchie(id_hierarchie) ON DELETE RESTRICT
);

CREATE TABLE Annonce(
   id_publication INT AUTO_INCREMENT,
   nom_publication VARCHAR(100) NOT NULL,
   date_publication DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
   description TEXT,
   quantite INT NOT NULL DEFAULT 1,
   id_categorie INT NULL,
   id_Utilisateur INT NOT NULL,
   PRIMARY KEY(id_publication),
   FOREIGN KEY(id_categorie) REFERENCES Categorie(id_categorie) ON DELETE SET NULL,
   FOREIGN KEY(id_Utilisateur) REFERENCES Profil(id_Utilisateur) ON DELETE CASCADE
);

CREATE TABLE Administrateur(
   id_Utilisateur INT,
   PRIMARY KEY(id_Utilisateur),
   FOREIGN KEY(id_Utilisateur) REFERENCES Profil(id_Utilisateur) ON DELETE CASCADE
);

CREATE TABLE Utilisateur(
   id_Utilisateur INT,
   id_site INT NULL,
   id_departement INT NOT NULL,
   PRIMARY KEY(id_Utilisateur),
   FOREIGN KEY(id_Utilisateur) REFERENCES Profil(id_Utilisateur) ON DELETE CASCADE,
   FOREIGN KEY(id_site) REFERENCES Site(id_site) ON DELETE SET NULL,
   FOREIGN KEY(id_departement) REFERENCES Departement(id_departement) ON DELETE RESTRICT
);

CREATE TABLE Token(
   id_token INT AUTO_INCREMENT,
   token_hash CHAR(64) NOT NULL UNIQUE,
   type ENUM('verification_email','reset_password','remember_me') NOT NULL,
   date_creation DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
   date_expiration DATETIME NOT NULL,
   date_utilisation DATETIME NULL,
   id_Utilisateur INT NOT NULL,
   PRIMARY KEY(id_token),
   FOREIGN KEY(id_Utilisateur) REFERENCES Profil(id_Utilisateur) ON DELETE CASCADE
);

CREATE TABLE Reserve(
   id_publication INT,
   date_reservation DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
   id_Utilisateur INT NOT NULL,
   PRIMARY KEY(id_publication),
   FOREIGN KEY(id_publication) REFERENCES Annonce(id_publication) ON DELETE CASCADE,
   FOREIGN KEY(id_Utilisateur) REFERENCES Profil(id_Utilisateur) ON DELETE CASCADE
);

CREATE INDEX idx_token_user_type ON Token(id_Utilisateur, type);
