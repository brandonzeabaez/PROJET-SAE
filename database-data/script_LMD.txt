USE sae30392_mysql;

CREATE TABLE IF NOT EXISTS Hierachie(
   id_hierarchie INT,
   role VARCHAR(50),
   CONSTRAINT pk_hierarchie_con PRIMARY KEY(id_hierarchie)
);

-- Rôles des utilisateurs (2 = administrateur, accès au panel /admin)
INSERT IGNORE INTO Hierachie(id_hierarchie, role) VALUES (1, 'utilisateur'), (2, 'admin');

CREATE TABLE IF NOT EXISTS Permission(
   id_permission INT,
   libelle VARCHAR(50) NOT NULL,
   CONSTRAINT pk_permission_con PRIMARY KEY(id_permission)
);

CREATE TABLE IF NOT EXISTS User(
   id_user INT AUTO_INCREMENT,
   email VARCHAR(50) UNIQUE NOT NULL,
   mot_de_passe VARCHAR(255) NOT NULL,
   nom CHAR(50) NOT NULL,
   prenom CHAR(50) NOT NULL,
   telephone CHAR(16),
   pays CHAR(30),
   id_hierarchie INT DEFAULT 1,
   reset_token VARCHAR(255) NULL,
   reset_expire DATETIME NULL,
   email_verifie BOOLEAN NOT NULL DEFAULT 0,
   verification_token VARCHAR(255) NULL,
   verification_expire DATETIME NULL,
   CONSTRAINT pk_user_con PRIMARY KEY(id_user),
   CONSTRAINT fk_user_to_hierarchie_con FOREIGN KEY(id_hierarchie) REFERENCES Hierachie(id_hierarchie)
);

CREATE TABLE IF NOT EXISTS Publication(
   id_publication INT,
   nom_publication VARCHAR(50),
   date_publication DATE,
   id_user INT NOT NULL,
   CONSTRAINT pk_publication_con PRIMARY KEY(id_publication),
   CONSTRAINT fk_publication_con FOREIGN KEY(id_user) REFERENCES User(id_user)
);

CREATE TABLE IF NOT EXISTS Possede(
   id_hierarchie INT,
   id_permission INT,
   CONSTRAINT pk_possede_con PRIMARY KEY(id_hierarchie, id_permission),
   CONSTRAINT pk_possede_to_hierarchie_con FOREIGN KEY(id_hierarchie) REFERENCES Hierachie(id_hierarchie),
   CONSTRAINT fk_possede_to_permission_con FOREIGN KEY(id_permission) REFERENCES Permission(id_permission)
);   

CREATE TABLE IF NOT EXISTS Reserve(
   id_user INT,
   id_publication INT,
   date_reservation DATE,
   CONSTRAINT pk_possede_con PRIMARY KEY(id_user, id_publication),
   CONSTRAINT fk_user_con FOREIGN KEY(id_user) REFERENCES User(id_user),
   FOREIGN KEY(id_publication) REFERENCES Publication(id_publication)
);

-- Comptes des départements de l'IUT, créés par un administrateur.
-- Le secrétariat reçoit un lien unique (jeton stocké haché en SHA-256) pour
-- saisir le nom du département, la ville et choisir son mot de passe.
-- mot_de_passe est NULL tant que cette première connexion n'est pas faite.
CREATE TABLE IF NOT EXISTS Departement(
   id_departement INT AUTO_INCREMENT,
   email_secretariat VARCHAR(100) UNIQUE NOT NULL,
   mot_de_passe VARCHAR(255) NULL,
   nom_departement VARCHAR(100) NULL,
   ville VARCHAR(100) NULL,
   actif BOOLEAN NOT NULL DEFAULT 0,
   setup_token CHAR(64) NULL,
   setup_expire DATETIME NULL,
   date_creation DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
   CONSTRAINT pk_departement_con PRIMARY KEY(id_departement)
);
